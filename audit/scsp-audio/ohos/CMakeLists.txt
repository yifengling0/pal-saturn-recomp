cmake_minimum_required(VERSION 3.16)
project(pal_ohos_production_runtime_objects LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
get_filename_component(PAL_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." REALPATH)
set(RT "${PAL_ROOT}/third_party/saturn-recomp-pal/runtime")
set(ENTRY "${PAL_ROOT}/harmony/pal-saturn-openharmony/entry/src/main/cpp")
set(SDL_TESTS OFF CACHE BOOL "" FORCE)
set(SDL_EXAMPLES OFF CACHE BOOL "" FORCE)
set(SDL_HAPTIC OFF CACHE BOOL "" FORCE)
set(SDL_SHARED ON CACHE BOOL "" FORCE)
set(SDL_STATIC OFF CACHE BOOL "" FORCE)
add_subdirectory("${PAL_ROOT}/third_party/sdl3-src" sdl3)
set(SDL3_DIR "${ENTRY}/cmake")
include("${RT}/runtime.cmake")
if(NOT SDL3_FOUND)
 message(FATAL_ERROR "Production SDL host must be built; null host is not allowed")
endif()
get_target_property(HOST_SOURCES saturn_host SOURCES)
message(STATUS "Actual production host sources: ${HOST_SOURCES}")
file(SHA256 "${RT}/src/host/host.cpp" HOST_SHA)
file(SHA256 "${ENTRY}/pal_ohos_audio.cpp" AUDIO_SHA)
file(SHA256 "${RT}/src/video/vdp2.cpp" VDP2_SHA)
file(SHA256 "${RT}/src/machine/machine.cpp" CPU_SHA)
target_compile_definitions(saturn_host PRIVATE
 PAL_OHOS_DEFAULTS=1 PAL_OHOS_INPUT=1 PAL_OHOS_GAMEPAD_BRIDGE=1 PAL_OHOS_AUDIO_PULL=1 PAL_OHOS_DISPLAY=1
 "PAL_HOST_UI_SOURCE_SHA256=\"${HOST_SHA}\""
 "PAL_OHOS_AUDIO_SOURCE_SHA256=\"${AUDIO_SHA}\"")
target_sources(saturn_host PRIVATE "${ENTRY}/pal_ohos_gamepad.cpp" "${ENTRY}/pal_ohos_audio.cpp")
target_include_directories(saturn_host PRIVATE "${ENTRY}")
target_compile_definitions(saturn_machine PRIVATE PAL_SERIAL_SLAVE_COROUTINE=1
 "PAL_VDP2_COMPOSE_SOURCE_SHA256=\"${VDP2_SHA}\""
 "PAL_CPU_HANDOFF_SOURCE_SHA256=\"${CPU_SHA}\"")
add_library(pal_display SHARED "${ENTRY}/pal_ohos_display.cpp")
target_include_directories(pal_display PRIVATE "${RT}/src/host")
target_link_libraries(pal_display PRIVATE libace_napi.z.so)
target_link_libraries(saturn_host PUBLIC pal_display ${CMAKE_DL_LIBS} libhilog_ndk.z.so)
# These independent source files contain no generated modules or asset proof definitions.
# They are only compiled to objects here, not linked into an application.
add_library(pal_game_independent OBJECT "${PAL_ROOT}/src/pal_game/pal_audio_stream_init.cpp" "${PAL_ROOT}/src/pal_game/pal_diag.cpp")
target_include_directories(pal_game_independent PRIVATE "${RT}/include" "${RT}/src/machine")
add_custom_target(production_runtime_objects DEPENDS saturn_core saturn_machine saturn_host saturn_m68k saturn_imgui pal_game_independent)
