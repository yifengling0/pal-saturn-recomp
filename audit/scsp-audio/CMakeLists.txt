cmake_minimum_required(VERSION 3.20)
project(pal_scsp_audio_audit C CXX)
set(CMAKE_CXX_STANDARD 20)
get_filename_component(PAL_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(RUNTIME "${PAL_ROOT}/third_party/saturn-recomp-pal/runtime" CACHE PATH "Actual runtime under test")
option(LEGACY_SCSP "Baseline lacks the new DSP carry state functions" OFF)
option(SANITIZE "ASan and UBSan" OFF)
set(INCLUDES "${RUNTIME}/include" "${RUNTIME}/src/machine" "${RUNTIME}/src/sound" "${RUNTIME}/src/host" "${RUNTIME}/src/video" "${RUNTIME}/tests")
if(SANITIZE)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
add_executable(scsp_regressions "${RUNTIME}/src/sound/scsp.cpp" "${RUNTIME}/tests/main.cpp" dsp_regression.cpp slot_regression.cpp unit_support.cpp)
target_include_directories(scsp_regressions PRIVATE ${INCLUDES})
target_compile_options(scsp_regressions PRIVATE -UNDEBUG -include "${CMAKE_CURRENT_LIST_DIR}/carry_compat.h")
if(LEGACY_SCSP)
  target_compile_definitions(scsp_regressions PRIVATE LEGACY_SCSP=1)
endif()
set(M68K "${RUNTIME}/third_party/musashi")
add_library(audit_m68k STATIC "${M68K}/m68kcpu.c" "${M68K}/m68kops.c" "${M68K}/m68kdasm.c" "${M68K}/softfloat/softfloat.c" "${RUNTIME}/src/sound/m68k_state.c")
target_include_directories(audit_m68k PUBLIC "${M68K}")
add_executable(sound_pipeline "${RUNTIME}/src/sound/scsp.cpp" "${RUNTIME}/src/sound/sound.cpp" "${RUNTIME}/src/machine/mmio.cpp" sound_pipeline.cpp)
target_include_directories(sound_pipeline PRIVATE ${INCLUDES})
target_compile_options(sound_pipeline PRIVATE -UNDEBUG -ffunction-sections -fdata-sections)
target_link_options(sound_pipeline PRIVATE -Wl,--gc-sections)
target_link_libraries(sound_pipeline PRIVATE audit_m68k)
if(NOT LEGACY_SCSP)
  add_executable(state_loader "${RUNTIME}/src/machine/state.cpp" "${RUNTIME}/src/sound/scsp.cpp" "${RUNTIME}/tests/no_modules.cpp" state_loader.cpp)
  target_include_directories(state_loader PRIVATE ${INCLUDES})
  target_compile_options(state_loader PRIVATE -UNDEBUG -ffunction-sections -fdata-sections)
  target_link_options(state_loader PRIVATE -Wl,--gc-sections)
endif()
