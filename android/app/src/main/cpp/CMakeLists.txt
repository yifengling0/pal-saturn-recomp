cmake_minimum_required(VERSION 3.22)
project(pal_android LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(NOT ANDROID OR NOT ANDROID_ABI STREQUAL "arm64-v8a")
  message(FATAL_ERROR "This port requires Android arm64-v8a NDK")
endif()
foreach(p PAL_ROOT PAL_RECOMP PAL_RUNTIME PAL_SDL)
  if(NOT IS_DIRECTORY "${${p}}")
    message(FATAL_ERROR "Missing ${p} directory")
  endif()
endforeach()
set(SDL_SHARED ON CACHE BOOL "" FORCE)
set(SDL_STATIC OFF CACHE BOOL "" FORCE)
set(SDL_TESTS OFF CACHE BOOL "" FORCE)
set(SDL_EXAMPLES OFF CACHE BOOL "" FORCE)
add_subdirectory("${PAL_SDL}" sdl)
# The generated runtime calls find_package: provide its build-tree config explicitly.
set(SDL3_DIR "${CMAKE_CURRENT_SOURCE_DIR}/cmake")
set(SATURN_RUNTIME "${PAL_RUNTIME}" CACHE PATH "" FORCE)
set(SATURN_NO_SDL OFF CACHE BOOL "" FORCE)
set(SATURN_EXTRA "${PAL_ROOT}/src/pal_game/pal_game.cmake" CACHE FILEPATH "" FORCE)
add_subdirectory("${PAL_RECOMP}" recomp)
get_target_property(host_sources saturn_host SOURCES)
if(NOT host_sources MATCHES "host.cpp" OR host_sources MATCHES "host_null.cpp")
  message(FATAL_ERROR "Android must use real SDL host, never a headless stub")
endif()
if(TARGET selftest)
  set_target_properties(selftest PROPERTIES EXCLUDE_FROM_ALL TRUE)
endif()
target_compile_definitions(saturn_machine PRIVATE PAL_SERIAL_SLAVE_COROUTINE=1)
# Generated Android target saturn is already SHARED OUTPUT_NAME main, includes
# PAL hook registration and paired-frame queue exactly once.
get_target_property(kind saturn TYPE)
if(NOT kind STREQUAL "SHARED_LIBRARY")
  message(FATAL_ERROR "Generated Android saturn must be shared libmain")
endif()
target_link_libraries(saturn SDL3::SDL3)
target_link_options(saturn PRIVATE "-Wl,--no-undefined" "-Wl,-z,max-page-size=16384")
