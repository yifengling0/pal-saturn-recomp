cmake_minimum_required(VERSION 3.20)
project(pal_patch_tests CXX)
set(CMAKE_CXX_STANDARD 20)
set(SATURN_NO_SDL ON CACHE BOOL "headless" FORCE)
get_filename_component(ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(RT "${ROOT}/third_party/saturn-recomp-pal/runtime")
include("${RT}/runtime.cmake")
add_executable(patch_core core_test.cpp)
target_link_libraries(patch_core saturn_core)
add_executable(patch_hardware hardware_test.cpp "${RT}/tests/no_modules.cpp")
target_link_libraries(patch_hardware saturn_core saturn_machine saturn_host Threads::Threads)
# Assertions are the test oracle even when invoked from a Release build.
target_compile_options(patch_core PRIVATE -UNDEBUG)
target_compile_options(patch_hardware PRIVATE -UNDEBUG)
add_executable(patch_dma dma_test.cpp "${RT}/tests/no_modules.cpp")
target_link_libraries(patch_dma saturn_core saturn_machine saturn_host Threads::Threads)
target_compile_options(patch_dma PRIVATE -UNDEBUG)
